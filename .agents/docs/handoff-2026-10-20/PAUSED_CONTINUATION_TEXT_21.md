void fail(uint64_t bit) { state->failures.fetch_or(bit, std::memory_order_relaxed); }
bool identifier(const char *text, size_t length) {
    if (!length || length >= NAME) return false;
    for (size_t i=0; i<length; ++i) {
        unsigned char c=static_cast<unsigned char>(text[i]);
        if (!((c>='a'&&c<='z') || (c>='A'&&c<='Z') || (c>='0'&&c<='9')
              || c=='_' || c=='$')) return false;
    }
    return true;
}
bool space(char c) { return c==' ' || c=='\n' || c=='\r' || c=='\t'; }
// Source-bound original NVRTC PTX only. Unknown/binary LoadData is refused.
bool classify_ptx(const char *text, size_t &length, uint32_t &target) {
    if (!text || !((text[0]=='/' && text[1]=='/') || text[0]=='.')) return false;
    length=strnlen(text, IMAGE);
    if (!length || length==IMAGE) return false;
    target=UINT32_MAX;
    uint32_t matches=0;
    for (size_t i=0; i<length;) {
        const unsigned char c=static_cast<unsigned char>(text[i]);
        if (c>=127 || (c<32 && !space(char(c)))) return false;
        if (i+1<length && text[i]=='/' && text[i+1]=='/') {
            while (i<length && text[i]!='\n') ++i;
            continue;
        }
        if (i+1<length && text[i]=='/' && text[i+1]=='*') {
            i+=2;
            while (i+1<length && !(text[i]=='*' && text[i+1]=='/')) ++i;
            if (i+1>=length) return false;
            i+=2; continue;
        }
        if (text[i]=='"') {
            ++i;
            while (i<length && text[i]!='"') {
                if (text[i]=='\\') ++i;
                ++i;
            }
            if (i>=length) return false;
            ++i; continue;
        }
        if (i+7<=length && !memcmp(text+i,".entry",6) && space(text[i+6])) {
            size_t begin=i+6;
            while (begin<length && space(text[begin])) ++begin;
            size_t end=begin;
            while (end<length && !space(text[end]) && text[end]!='(') ++end;
            if (!identifier(text+begin,end-begin)) return false;
            for (uint32_t k=0; k<state->target_count; ++k) {
                size_t n=strlen(state->targets[k]);
                if (n==end-begin && !memcmp(text+begin,state->targets[k],n)) {
                    target=k; ++matches;
                }
            }
            i=end; continue;
        }
        ++i;
    }
    return matches<=1;
}
uint32_t find_module(uintptr_t handle) {
    uint32_t found=UINT32_MAX;
    uint32_t count=state->module_count.load(std::memory_order_acquire);
    for (uint32_t i=0; i<count && i<MODULES; ++i) {
        Module &m=state->modules[i];
        if (m.loaded.load(std::memory_order_acquire) && !m.retired.load() && m.handle==handle) {
            if (found!=UINT32_MAX) { fail(AMBIGUOUS); return UINT32_MAX; }
            found=i;
        }
    }
    return found;
}
uint32_t find_function(uintptr_t handle) {
    uint32_t found=UINT32_MAX;
    uint32_t count=state->function_count.load(std::memory_order_acquire);
    for (uint32_t i=0; i<count && i<FUNCTIONS; ++i) {
        Function &f=state->functions[i];
        if (f.ready.load(std::memory_order_acquire) && f.handle==handle &&
            !state->modules[f.module].retired.load()) {
            if (found!=UINT32_MAX) { fail(AMBIGUOUS); return UINT32_MAX; }
            found=i;
        }
    }
    return found;
}
void event(uint32_t kind, uint32_t site, const CUpti_CallbackData *cb,
           uint32_t module, uint32_t function, int32_t status,
           const cuLaunchKernel_params *launch=nullptr) {
    uint32_t i=state->event_count.fetch_add(1,std::memory_order_relaxed);
    if (i>=EVENTS) { fail(CAPACITY); return; }
    Event &e=state->events[i];
    e.sequence=uint64_t(i)+1; e.tid=uint64_t(syscall(SYS_gettid));
    e.kind=kind; e.site=site; e.module=module; e.function=function; e.status=status;
    e.context_uid=cb ? cb->contextUid : state->modules[module].context_uid;
    e.correlation=cb ? cb->correlationId : frame.correlation;
    if (module<MODULES) e.module_handle=state->modules[module].handle;
    if (function<FUNCTIONS) e.function_handle=state->functions[function].handle;
    if (launch) {
        e.function_handle=reinterpret_cast<uintptr_t>(launch->f);
        e.stream=reinterpret_cast<uintptr_t>(launch->hStream);
        e.grid[0]=launch->gridDimX; e.grid[1]=launch->gridDimY; e.grid[2]=launch->gridDimZ;
        e.block[0]=launch->blockDimX; e.block[1]=launch->blockDimY; e.block[2]=launch->blockDimZ;
        e.shared=launch->sharedMemBytes;
    }
}
void driver_enter(CUpti_CallbackId id, const CUpti_CallbackData *cb) {
    if (frame.kind) { fail(REENTRANT); return; }
    frame.correlation=cb->correlationId;
    if (id==CUPTI_DRIVER_TRACE_CBID_cuModuleLoadData) {
        const auto *p=static_cast<const cuModuleLoadData_params *>(cb->functionParams);
        frame.kind=LOAD;
        if (!p || !p->module || !p->image || !cb->context || !cb->contextUid) { fail(ABI); return; }
        size_t length=0; uint32_t target=UINT32_MAX;
        if (!classify_ptx(static_cast<const char *>(p->image),length,target)) { fail(FOREIGN); return; }
        uint32_t i=state->module_count.fetch_add(1,std::memory_order_relaxed);
        if (i>=MODULES) { fail(CAPACITY); return; }
        frame.module=i; frame.output=reinterpret_cast<uintptr_t>(p->module);
        Module &m=state->modules[i];
        m.context=reinterpret_cast<uintptr_t>(cb->context); m.context_uid=cb->contextUid;
        m.correlation=cb->correlationId; m.target=target; m.ptx_bytes=length;
        if (target!=UINT32_MAX) {
            uint32_t payload=state->payload_count.fetch_add(1,std::memory_order_relaxed);
            if (payload>=PAYLOADS) { fail(CAPACITY); return; }
            m.payload=payload;
            memcpy(state->payloads[payload].ptx,p->image,length);
            event(LOAD,CUPTI_API_ENTER,cb,i,UINT32_MAX,0);
        }
    } else if (id==CUPTI_DRIVER_TRACE_CBID_cuModuleGetFunction) {
        const auto *p=static_cast<const cuModuleGetFunction_params *>(cb->functionParams);
        frame.kind=GET_FUNCTION;
        if (!p || !p->hfunc || !p->hmod || !p->name) { fail(ABI); return; }
        uint32_t i=find_module(reinterpret_cast<uintptr_t>(p->hmod));
        if (i==UINT32_MAX) { fail(FOREIGN); return; }
        frame.module=i; frame.output=reinterpret_cast<uintptr_t>(p->hfunc);
        if (state->modules[i].target==UINT32_MAX) { state->ignored_functions.fetch_add(1); return; }
        size_t n=strnlen(p->name,NAME);
        if (!identifier(p->name,n) || strcmp(p->name,state->targets[state->modules[i].target])) {
            fail(AMBIGUOUS); return;
        }
        uint32_t f=state->function_count.fetch_add(1,std::memory_order_relaxed);
        if (f>=FUNCTIONS) { fail(CAPACITY); return; }
        frame.function=f; state->functions[f].module=i;
        memcpy(state->functions[f].name,p->name,n+1);
        event(GET_FUNCTION,CUPTI_API_ENTER,cb,i,f,0);
    } else if (id==CUPTI_DRIVER_TRACE_CBID_cuLaunchKernel) {
        const auto *p=static_cast<const cuLaunchKernel_params *>(cb->functionParams);
        frame.kind=LAUNCH;
        if (!p || !p->f) { fail(ABI); return; }
        uint32_t f=find_function(reinterpret_cast<uintptr_t>(p->f));
        frame.function=f;
        if (f==UINT32_MAX) { state->ignored_launches.fetch_add(1); return; }
        frame.module=state->functions[f].module;
        Module &m=state->modules[frame.module];
        if (cb->contextUid!=m.context_uid || reinterpret_cast<uintptr_t>(cb->context)!=m.context) {
            fail(FOREIGN); return;
        }
        if (!p->gridDimX || !p->gridDimY || !p->gridDimZ || !p->blockDimX || !p->blockDimY || !p->blockDimZ) {
            fail(ABI); return;
        }
        event(LAUNCH,CUPTI_API_ENTER,cb,frame.module,f,0,p);
    } else if (id==CUPTI_DRIVER_TRACE_CBID_cuModuleUnload) {
        const auto *p=static_cast<const cuModuleUnload_params *>(cb->functionParams);
        frame.kind=UNLOAD;
        if (!p || !p->hmod) { fail(ABI); return; }
        frame.module=find_module(reinterpret_cast<uintptr_t>(p->hmod));
        if (frame.module==UINT32_MAX) { fail(FOREIGN); return; }
        if (state->modules[frame.module].target!=UINT32_MAX)
            event(UNLOAD,CUPTI_API_ENTER,cb,frame.module,UINT32_MAX,0);
    } else { fail(ABI); }
}
void driver_exit(CUpti_CallbackId id, const CUpti_CallbackData *cb) {
    uint32_t wanted=id==CUPTI_DRIVER_TRACE_CBID_cuModuleLoadData ? LOAD :
        id==CUPTI_DRIVER_TRACE_CBID_cuModuleGetFunction ? GET_FUNCTION :
        id==CUPTI_DRIVER_TRACE_CBID_cuLaunchKernel ? LAUNCH : UNLOAD;
    if (frame.kind!=wanted || frame.correlation!=cb->correlationId || !cb->functionReturnValue) {
        fail(ASYNC); frame={}; return;
    }
    int32_t result=int32_t(*static_cast<const CUresult *>(cb->functionReturnValue));
    if (result!=CUDA_SUCCESS) fail(NATIVE_ERROR);
    if (frame.module<MODULES) {
        Module &m=state->modules[frame.module];
        if (cb->contextUid!=m.context_uid || reinterpret_cast<uintptr_t>(cb->context)!=m.context) fail(FOREIGN);
        if (wanted==LOAD && result==CUDA_SUCCESS) {
            const auto *p=static_cast<const cuModuleLoadData_params *>(cb->functionParams);
            if (!p || reinterpret_cast<uintptr_t>(p->module)!=frame.output || !*p->module || m.resources!=1) fail(INCOMPLETE);
            else { m.handle=reinterpret_cast<uintptr_t>(*p->module); m.loaded.store(true,std::memory_order_release); }
        } else if (wanted==GET_FUNCTION && frame.function<FUNCTIONS && result==CUDA_SUCCESS) {
            const auto *p=static_cast<const cuModuleGetFunction_params *>(cb->functionParams);
            if (!p || reinterpret_cast<uintptr_t>(p->hfunc)!=frame.output || !*p->hfunc ||
                reinterpret_cast<uintptr_t>(p->hmod)!=m.handle) fail(ABI);
            else { Function &f=state->functions[frame.function]; f.handle=reinterpret_cast<uintptr_t>(*p->hfunc);
                   f.ready.store(true,std::memory_order_release); }
        } else if (wanted==UNLOAD && result==CUDA_SUCCESS) m.retired.store(true,std::memory_order_release);
        if (m.target!=UINT32_MAX) event(wanted,CUPTI_API_EXIT,cb,frame.module,frame.function,result,
            wanted==LAUNCH ? static_cast<const cuLaunchKernel_params *>(cb->functionParams) : nullptr);
    }
    frame={};
}
void resource(CUpti_CallbackId id, const void *data) {
    // Exact NVIDIA12.9.79 sample: cbdata wrapper -> descriptor -> module data.
    if (!data || frame.module>=MODULES) { fail(ASYNC); return; }
    const auto *wrapper=static_cast<const CUpti_ResourceData *>(data);
    const auto *module=static_cast<const CUpti_ModuleResourceData *>(wrapper->resourceDescriptor);
    if (!module) { fail(ABI); return; }
    Module &m=state->modules[frame.module];
    if (id==CUPTI_CBID_RESOURCE_MODULE_LOADED) {
        if (frame.kind!=LOAD || ++m.resources!=1) { fail(AMBIGUOUS); return; }
        m.module_id=module->moduleId;
        if (m.target!=UINT32_MAX) {
            if (m.payload>=PAYLOADS || !module->pCubin || !module->cubinSize || module->cubinSize>IMAGE) { fail(BOUNDS); return; }
            m.cubin_bytes=module->cubinSize;
            memcpy(state->payloads[m.payload].cubin,module->pCubin,module->cubinSize);
            event(RESOURCE_LOAD,0,nullptr,frame.module,UINT32_MAX,0);
        }
    } else if (id==CUPTI_CBID_RESOURCE_MODULE_UNLOAD_STARTING) {
        if (frame.kind!=UNLOAD || module->moduleId!=m.module_id) { fail(ASYNC); return; }
        if (m.target!=UINT32_MAX) event(RESOURCE_UNLOAD,0,nullptr,frame.module,UINT32_MAX,0);
    } else fail(ABI);
}
void CUPTIAPI callback(void *, CUpti_CallbackDomain domain, CUpti_CallbackId id, const void *data) {
    int saved=errno;
    state->active.fetch_add(1,std::memory_order_acq_rel);
    if (!state->closing.load(std::memory_order_acquire)) {
        if (domain==CUPTI_CB_DOMAIN_RESOURCE) resource(id,data);
        else if (domain==CUPTI_CB_DOMAIN_DRIVER_API && data) {
            const auto *cb=static_cast<const CUpti_CallbackData *>(data);
            if (cb->callbackSite==CUPTI_API_ENTER) driver_enter(id,cb);
            else if (cb->callbackSite==CUPTI_API_EXIT) driver_exit(id,cb);
            else fail(ABI);
        } else fail(ABI);
    }
    state->active.fetch_sub(1,std::memory_order_acq_rel);
    errno=saved;
}

bool write_all(int fd, const void *raw, size_t n) {
    const char *p=static_cast<const char *>(raw);
    while (n) {
        ssize_t used=write(fd,p,n);
        if (used<0 && errno==EINTR) continue;
        if (used<=0) return false;
        p+=used; n-=size_t(used);
    }
    return true;
}
bool output(const char *name, const void *data, size_t n) {
    int fd=openat(state->directory,name,O_WRONLY|O_CREAT|O_EXCL|O_NOFOLLOW|O_CLOEXEC,0600);
    if (fd<0) return false;
    bool good=write_all(fd,data,n) && fsync(fd)==0;
    if (close(fd)) good=false;
    return good;
}
uint64_t now() {
    timespec t{};
    if (clock_gettime(CLOCK_MONOTONIC,&t)) return 0;
    return uint64_t(t.tv_sec)*1000000000ULL+uint64_t(t.tv_nsec);
}
void finish() {
    // No CUDA/CUPTI call, unsubscribe/finalize, driver destruction or arena free.
    // Gate every later callback before mutation; retain its userdata to OS exit.
    state->closing.store(true,std::memory_order_release);
    uint64_t begin=now();
    while (state->active.load(std::memory_order_acquire)) {
        uint64_t at=now();
        if (!begin || !at || at<begin || at-begin>5000000000ULL) { fail(DRAIN); return; }
        timespec wait{0,1000000}; nanosleep(&wait,nullptr);
    }
    uint32_t modules=state->module_count.load(), payloads=state->payload_count.load();
    uint32_t functions=state->function_count.load(), events=state->event_count.load();
    bool good=modules<=MODULES && payloads<=PAYLOADS && functions<=FUNCTIONS && events<=EVENTS;
    for (uint32_t i=0; i<modules && i<MODULES; ++i) {
        Module &m=state->modules[i];
        if (m.target==UINT32_MAX) continue;
        if (!m.loaded.load() || m.resources!=1 || !m.ptx_bytes || !m.cubin_bytes || m.payload>=PAYLOADS) { good=false; continue; }
        char name[64];
        snprintf(name,sizeof(name),"module-%04u.ptx",i);
        good=output(name,state->payloads[m.payload].ptx,m.ptx_bytes) && good;
        snprintf(name,sizeof(name),"module-%04u.cubin",i);
        good=output(name,state->payloads[m.payload].cubin,m.cubin_bytes) && good;
    }
    int fd=openat(state->directory,"capture.jsonl",O_WRONLY|O_CREAT|O_EXCL|O_NOFOLLOW|O_CLOEXEC,0600);
    if (fd<0) good=false;
    char row[2048];
    auto emit=[&](int n) { if (n<=0 || size_t(n)>=sizeof(row) || fd<0 || !write_all(fd,row,size_t(n))) good=false; };
    for (uint32_t i=0; i<modules && i<MODULES; ++i) {
        Module &m=state->modules[i]; if (m.target==UINT32_MAX) continue;
        emit(snprintf(row,sizeof(row),"{\"type\":\"module\",\"index\":%u,\"target\":\"%s\",\"module_id\":%u,\"module_handle\":%llu,\"context\":%llu,\"context_uid\":%u,\"correlation\":%u,\"resource_count\":%u,\"loaded\":%s,\"retired\":%s,\"ptx_bytes\":%zu,\"cubin_bytes\":%zu}\n",i,state->targets[m.target],m.module_id,(unsigned long long)m.handle,(unsigned long long)m.context,m.context_uid,m.correlation,m.resources,m.loaded.load()?"true":"false",m.retired.load()?"true":"false",m.ptx_bytes,m.cubin_bytes));
    }
    for (uint32_t i=0; i<functions && i<FUNCTIONS; ++i) {
        Function &f=state->functions[i];
        emit(snprintf(row,sizeof(row),"{\"type\":\"function\",\"index\":%u,\"module\":%u,\"name\":\"%s\",\"handle\":%llu,\"ready\":%s}\n",i,f.module,f.name,(unsigned long long)f.handle,f.ready.load()?"true":"false"));
    }
    for (uint32_t i=0; i<events && i<EVENTS; ++i) {
        const Event &e=state->events[i];
        emit(snprintf(row,sizeof(row),"{\"type\":\"event\",\"sequence\":%llu,\"tid\":%llu,\"kind\":%u,\"site\":%u,\"module\":%u,\"function\":%u,\"module_handle\":%llu,\"function_handle\":%llu,\"context_uid\":%u,\"correlation\":%u,\"status\":%d,\"stream\":%llu,\"grid\":[%u,%u,%u],\"block\":[%u,%u,%u],\"shared\":%u}\n",(unsigned long long)e.sequence,(unsigned long long)e.tid,e.kind,e.site,e.module,e.function,(unsigned long long)e.module_handle,(unsigned long long)e.function_handle,e.context_uid,e.correlation,e.status,(unsigned long long)e.stream,e.grid[0],e.grid[1],e.grid[2],e.block[0],e.block[1],e.block[2],e.shared));
    }
    if (fd>=0) { if (fsync(fd)) good=false; if (close(fd)) good=false; }
    if (!good) fail(EVIDENCE);
    int n=snprintf(row,sizeof(row),"{\"schema\":\"zg.loaded-sass.callback-only.v3\",\"status\":\"CAPTURE_ONLY_NOT_READY_OR_SASS_QUALIFIED\",\"nonce\":\"%s\",\"pid\":%d,\"parent_pid\":%d,\"arena_bytes\":%zu,\"actual_state_bytes\":%zu,\"module_metadata_count\":%u,\"selected_module_count\":%u,\"function_count\":%u,\"event_count\":%u,\"ignored_functions\":%llu,\"ignored_launches\":%llu,\"failures\":%llu,\"active_callbacks\":%llu,\"activity_api_enabled\":false,\"observer_CUDA_calls\":0,\"observer_CUPTI_calls_at_finish\":0,\"userdata_and_DSO_retained_to_OS_exit\":true}\n",state->nonce,state->pid,state->parent,ARENA,sizeof(State),modules,payloads,functions,events,(unsigned long long)state->ignored_functions.load(),(unsigned long long)state->ignored_launches.load(),(unsigned long long)state->failures.load(),(unsigned long long)state->active.load());
    if (n>0 && size_t(n)<sizeof(row)) (void)output("collector-finish.json",row,size_t(n));
}
int fd_env(const char *key) {
    const char *v=getenv(key); char *end=nullptr;
    if (!v || !*v) return -1;
    errno=0; long n=strtol(v,&end,10);
    return errno || !end || *end || n<3 || n>1048576 ? -1 : int(n);
}
bool same_file(const struct stat &a, const struct stat &b) {
    return S_ISREG(a.st_mode) && S_ISREG(b.st_mode) && a.st_dev==b.st_dev && a.st_ino==b.st_ino && a.st_size==b.st_size;
}
template<class T> bool symbol(T &out,const char *name,const struct stat &wanted) {
    static_assert(sizeof(T)==sizeof(void *));
    void *raw=dlsym(state->cupti,name); Dl_info info{}; struct stat actual{};
    if (!raw || !dladdr(raw,&info) || !info.dli_fname || stat(info.dli_fname,&actual) || !same_file(wanted,actual)) return false;
    memcpy(&out,&raw,sizeof(out)); return true;
}
bool initialise() {
    int cupti_fd=fd_env("ZG_SASS_CUPTI_FD");
    state->directory=fd_env("ZG_SASS_DIRECTORY_FD");
    const char *nonce=getenv("ZG_SASS_NONCE"), *targets=getenv("ZG_SASS_ENTRYPOINTS");
    struct stat directory{}, library{};
    if (cupti_fd<0 || state->directory<0 || !nonce || strlen(nonce)!=32 || strspn(nonce,"0123456789abcdef")!=32 ||
        !targets || !*targets || strlen(targets)>=PAYLOADS*NAME || fstat(state->directory,&directory) ||
        !S_ISDIR(directory.st_mode) || directory.st_uid!=geteuid() || (directory.st_mode&0777)!=0700 ||
        fstat(cupti_fd,&library) || !S_ISREG(library.st_mode) || library.st_size!=7604800) return false;
    strcpy(state->nonce,nonce); state->pid=getpid(); state->parent=getppid();
    const char *at=targets;
    while (*at) {
        const char *end=strchr(at,','); size_t n=end ? size_t(end-at) : strlen(at);
        if (state->target_count==PAYLOADS || !identifier(at,n)) return false;
        for (uint32_t i=0; i<state->target_count; ++i) if (strlen(state->targets[i])==n && !memcmp(at,state->targets[i],n)) return false;
        memcpy(state->targets[state->target_count++],at,n);
        if (!end) break;
        at=end+1; if (!*at) return false;
    }
    char path[64]; snprintf(path,sizeof(path),"/proc/self/fd/%d",cupti_fd);
    state->cupti=dlopen(path,RTLD_NOW|RTLD_LOCAL);
    if (!state->cupti || !symbol(state->api.version,"cuptiGetVersion",library) ||
